% =========================================================
% LAYOUT 2: LABEL AND VALUE ROWS
% =========================================================

% The left column automatically takes the width required by
% the longest label in the current block.
%
% \CVSkillGap controls the distance between the label and
% value columns.

\newcommand{\CVLabelRows}[1]{%
  \ifblank{#1}
    {}
    {%
      \begin{tabularx}{\textwidth}{%
        @{}
        >{\bfseries}l
        @{\hspace{\CVSkillGap}}
        X
        @{}
      }
        #1%
      \end{tabularx}
      \par
    }%
}

% Arguments:
% 1: Label
% 2: Value

\newcommand{\CVLabelRow}[2]{%
  \ifblank{#1#2}
    {}
    {%
      #1 & #2 \\[\CVSkillRowSpace]%
    }%
}

% =========================================================

\newcommand{\CVEntries}[1]{%
  \ifblank{#1}
    {}
    {%
      \begin{itemize}[
        leftmargin=0.15in,
        label={},
        itemsep=0pt,
        topsep=0pt,
        parsep=0pt,
        partopsep=0pt
      ]
        #1%
      \end{itemize}
    }%
}

% =========================================================
% LAYOUT 3: UNIVERSAL ENTRY
% =========================================================

\newcommand{\CVEntrySeparator}{%
  \enspace$|$\enspace
}

% Arguments:
%
% 1: Main title
% 2: Secondary left-side text
% 3: URL
% 4: Font Awesome icon name
% 5: Organization or location
% 6: Date
% 7: Full-width body
%
% Empty arguments are omitted automatically.
% Separators are inserted only between populated fields.

\newcommand{\CVEntry}[7]{%
  \ifblank{#1#2#3#5#6#7}
    {}
    {%
      \vspace{\CVEntryBeforeSpace}
      \item

      \ifblank{#1#2#3#5#6}
        {}
        {%
          \begin{tabular*}{0.97\textwidth}[t]{%
            @{}l@{\extracolsep{\fill}}r@{}
          }

            % Top-left content

            \ifblank{#1}
              {}
              {\textbf{#1}}%

            \ifblank{#2}
              {}
              {%
                \ifblank{#1}
                  {}
                  {\CVEntrySeparator}%
                #2%
              }%

            \ifblank{#3}
              {}
              {%
                \ifblank{#1#2}
                  {}
                  {\CVEntrySeparator}%
                \href{#3}{%
                  \color{blue}%
                  \ifblank{#4}
                    {\faIcon{link}}
                    {\faIcon{#4}}%
                }%
              }%

            &

            % Top-right content

            \ifblank{#5}
              {}
              {#5}%

            \ifblank{#6}
              {}
              {%
                \ifblank{#5}
                  {}
                  {\CVEntrySeparator}%
                #6%
              }%

            \\%
          \end{tabular*}%
        }%
      
      \CVSmallGap
      
      \ifblank{#7}
        {}
        {#7}%

      \vspace{\CVEntryAfterSpace}
    }%
}

% =========================================================
% UNIVERSAL ENTRY CONTENT
% =========================================================

% Optional argument:
% true  = show bullet, default
% false = no bullet
%
% Examples:
%
% \CVContent{
%   \textbf{Technologies:} Python, TensorFlow, and R
% }
%
% \CVContent[false]{
%   Supporting text without a bullet.
% }

\newcommand{\CVContent}[2][true]{%
  \ifblank{#2}
    {}
    {%
      \ifstrequal{#1}{false}
        {%
          \noindent
          \hspace{0.15in}%
          {\small #2\par}%
          \vspace{\CVContentAfterSpace}%
        }
        {%
          \begin{itemize}[
            leftmargin=0.3in,
            label=\textbullet,
            itemsep=0pt,
            topsep=0pt,
            parsep=0pt,
            partopsep=0pt
          ]
            \item
            \small
            #2%
          \end{itemize}
          \vspace{\CVContentAfterSpace}%
        }%
    }%
}
